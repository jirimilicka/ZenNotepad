cmake_minimum_required(VERSION 3.16)
project(zen-notepad LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8)

add_executable(zen-notepad src/main.cpp src/editor.cpp src/document.cpp src/search.cpp src/portal.cpp src/translations.cpp)
target_link_libraries(zen-notepad PRIVATE Qt6::Widgets Qt6::DBus PkgConfig::PCRE2)
target_link_options(zen-notepad PRIVATE -Wl,-O1 -Wl,--as-needed)

include(GNUInstallDirs)
set(APP_ID io.github.jirimilicka.zen_notepad)
install(TARGETS zen-notepad RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/${APP_ID}.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/${APP_ID}.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES data/${APP_ID}.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)

option(BUILD_TESTING "Build tests" ON)
if(BUILD_TESTING)
  find_package(Qt6 COMPONENTS Test)
endif()
if(BUILD_TESTING AND Qt6Test_FOUND)
  add_executable(zen_test tests/test.cpp src/editor.cpp src/document.cpp src/search.cpp)
  target_link_libraries(zen_test PRIVATE Qt6::Widgets Qt6::Test PkgConfig::PCRE2)
endif()
if(BUILD_TESTING AND Qt6Test_FOUND)
  add_executable(zen_fuzz tests/fuzz.cpp src/editor.cpp src/document.cpp src/search.cpp src/portal.cpp src/translations.cpp)
  target_link_libraries(zen_fuzz PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test PkgConfig::PCRE2)
endif()
